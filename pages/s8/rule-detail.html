规则定义（DSL）

YAML
id: R-N01
name: 主键非空校验
category: 完整性
severity: P0
version: 3.2
owner: 张磊, 李婷

trigger:
  type: schedule | stream
  schedule: "0 */5 * * * *"   # 每 5 分钟
  stream: kafka.dq.events

scope:
  - field: order_id
    in: ["dwd_trade_order_di", "dwd_trade_refund_di", ...]
  - field: user_id
    in: ["dwd_user_register_di", ...]

condition: |
  SELECT COUNT(*) AS null_cnt
  FROM <scope_table>
  WHERE <pk_field> IS NULL
    OR TRIM(<pk_field>) = ''

threshold:
  fail_if_null_cnt_gt: 0

actions:
  on_fail:
    - emit_anomaly: severity=P0
    - block_writing: strategy=card_point
    - notify: [owner, dba]
  on_pass:
    - update_health_score: +1
